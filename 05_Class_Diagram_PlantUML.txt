STEP E – CLASS DIAGRAM

Paste the following PlantUML into any PlantUML-compatible editor. It can also be recreated in draw.io.

@startuml

class Movie {
  - title : string
  - language : string
  - durationMinutes : int
  + Movie(title, language, duration)
  + getTitle() : string
  + getLanguage() : string
  + getDuration() : int
}

enum SeatType {
  SILVER
  GOLD
  PLATINUM
}

enum SeatStatus {
  AVAILABLE
  BOOKED
}

class Seat {
  - number : int
  - type : SeatType
  + Seat(number, type)
  + getNumber() : int
  + getType() : SeatType
  + getPrice() : double
}

class Screen {
  - screenNumber : int
  - seats : vector<Seat>
  + Screen(screenNumber)
  + addSeat(seat)
  + getSeats() : vector<Seat>&
  + getScreenNumber() : int
}

class Cinema {
  - name : string
  - screens : vector<Screen>
  + Cinema(name)
  + addScreen(screen)
  + getScreens() : vector<Screen>&
  + getName() : string
}

class ShowSeat {
  - seat : Seat*
  - status : SeatStatus
  + ShowSeat(seat)
  + isAvailable() : bool
  + book() : bool
  + cancel() : void
  + getSeat() : Seat*
}

class Show {
  - movie : Movie*
  - screen : Screen*
  - startTime : string
  - showSeats : vector<ShowSeat>
  + Show(movie, screen, time)
  + displaySeats() : void
  + getMovie() : Movie*
  + getScreen() : Screen*
  + getStartTime() : string
  + getShowSeats() : vector<ShowSeat>&
}

class Customer {
  - name : string
  - phone : string
  + Customer(name, phone)
  + getName() : string
  + getPhone() : string
}

enum BookingStatus {
  PENDING
  CONFIRMED
  CANCELLED
}

class Booking {
  - static nextBookingId : int
  - bookingId : int
  - customer : Customer*
  - show : Show*
  - selectedSeats : vector<ShowSeat*>
  - totalAmount : double
  - status : BookingStatus
  + Booking(customer, show, seats, amount)
  + getBookingId() : int
  + getTotalAmount() : double
  + confirm() : void
  + cancel() : void
  + isConfirmed() : bool
}

abstract class Payment {
  + pay(amount : double) : bool
}

class UpiPayment {
  - success : bool
  + UpiPayment(success)
  + pay(amount) : bool
}

class CardPayment {
  - success : bool
  + CardPayment(success)
  + pay(amount) : bool
}

class CashPayment {
  - success : bool
  + CashPayment(success)
  + pay(amount) : bool
}

class PriceCalculator {
  + calculate(seats) : double
}

class TicketPrinter {
  + printTicket(booking) : void
}

class BookingService {
  + bookTicket(customer, show, seatNumbers, payment) : Booking*
  + cancelBooking(booking) : void
}

Cinema "1" *-- "1..*" Screen
Screen "1" *-- "1..*" Seat
Show "1" o-- "1" Movie
Show "1" --> "1" Screen
Show "1" *-- "1..*" ShowSeat
ShowSeat "*" --> "1" Seat
Booking "*" --> "1" Customer
Booking "*" o-- "1..*" ShowSeat
Booking "*" --> "1" Show
Booking "*" --> "1" Payment
Payment <|-- UpiPayment
Payment <|-- CardPayment
Payment <|-- CashPayment
BookingService --> Booking
BookingService --> PriceCalculator
BookingService --> TicketPrinter

@enduml
